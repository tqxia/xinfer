cmake_minimum_required(VERSION 3.24)
project(qwen_weight_memory LANGUAGES CXX)

find_package(Torch 2.13 REQUIRED)
find_package(CUDAToolkit REQUIRED)
set(XINFER_RUNTIME_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." CACHE PATH "Runtime headers to measure")
find_path(NLOHMANN_JSON_INCLUDE_DIR nlohmann/json.hpp REQUIRED)
find_library(XINFER_ATTENTION_OPS_LIBRARY xinfer_attention_ops REQUIRED)
get_target_property(PROBE_CXX_STANDARD torch CXX_STANDARD)
separate_arguments(PROBE_TORCH_FLAGS NATIVE_COMMAND "${TORCH_CXX_FLAGS}")
add_executable(qwen_weight_memory weight_memory.cpp)
target_compile_features(qwen_weight_memory PRIVATE cxx_std_${PROBE_CXX_STANDARD})
target_compile_options(qwen_weight_memory PRIVATE ${PROBE_TORCH_FLAGS})
target_include_directories(qwen_weight_memory PRIVATE "${XINFER_RUNTIME_ROOT}" "${NLOHMANN_JSON_INCLUDE_DIR}")
target_link_libraries(qwen_weight_memory PRIVATE ${TORCH_LIBRARIES} CUDA::cudart "${XINFER_ATTENTION_OPS_LIBRARY}")
