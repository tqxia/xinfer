# Header-only JSON parsing; pinned release and checksum, no runtime dependency.
include(FetchContent)
FetchContent_Declare(nlohmann_json
  URL https://github.com/nlohmann/json/releases/download/v3.11.3/json.tar.xz
  URL_HASH SHA256=d6c65aca6b1ed68e7a182f4757257b107ae403032760ed6ef121c9d55e81757d)
FetchContent_MakeAvailable(nlohmann_json)

option(QWEN_ENABLE_FLASHINFER "Build the experimental FlashInfer decode operator" OFF)
if(QWEN_ENABLE_FLASHINFER)
  set(XINFER_FLASHINFER_SOURCE_DIR "${PROJECT_SOURCE_DIR}/work/deps/flashinfer" CACHE PATH "Pinned FlashInfer checkout")
  set(FLASHINFER_REVISION beabf5004585b03da60e927bbb803acbaa62fe36)
  find_package(Git REQUIRED)
  execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${XINFER_FLASHINFER_SOURCE_DIR}" rev-parse HEAD
    OUTPUT_VARIABLE actual_revision OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
  if(NOT actual_revision STREQUAL FLASHINFER_REVISION)
    message(FATAL_ERROR "FlashInfer source must be v0.7.2.post1 at ${FLASHINFER_REVISION}")
  endif()
  execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${XINFER_FLASHINFER_SOURCE_DIR}" diff --exit-code HEAD -- include
    COMMAND_ERROR_IS_FATAL ANY)
  set(cccl "${XINFER_FLASHINFER_SOURCE_DIR}/3rdparty/cccl")
  execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${cccl}" rev-parse HEAD
    OUTPUT_VARIABLE cccl_revision OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
  if(NOT cccl_revision STREQUAL "16bd510c9b712e82b0ab6cbb630d8e29ba1f7116")
    message(FATAL_ERROR "Run bash tools/setup/fetch_flashinfer.sh to obtain the pinned CCCL headers")
  endif()
  execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${cccl}" diff --exit-code HEAD
    COMMAND_ERROR_IS_FATAL ANY)
  enable_language(CUDA)
  find_package(CUDAToolkit 13.0 REQUIRED)
  target_sources(xinfer_attention_ops PRIVATE custom_ops/flashinfer_decode.cu)
  target_include_directories(xinfer_attention_ops SYSTEM PRIVATE "${XINFER_FLASHINFER_SOURCE_DIR}/include")
  target_include_directories(xinfer_attention_ops SYSTEM BEFORE PRIVATE
    "${cccl}/libcudacxx/include" "${cccl}/cub" "${cccl}/thrust")
  target_compile_definitions(xinfer_attention_ops PRIVATE XINFER_FLASHINFER_REVISION="${FLASHINFER_REVISION}")
  target_compile_options(xinfer_attention_ops PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--expt-relaxed-constexpr>)
  set_target_properties(xinfer_attention_ops PROPERTIES CUDA_STANDARD ${QWEN_CXX_STANDARD}
    CUDA_STANDARD_REQUIRED ON CUDA_ARCHITECTURES 121)
  target_link_libraries(xinfer_attention_ops PRIVATE CUDA::cudart)
  configure_file("${XINFER_FLASHINFER_SOURCE_DIR}/LICENSE" "${CMAKE_BINARY_DIR}/FlashInfer-LICENSE" COPYONLY)
  configure_file("${cccl}/LICENSE" "${CMAKE_BINARY_DIR}/CCCL-LICENSE" COPYONLY)
endif()

endforeach()
