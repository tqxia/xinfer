find_package(Torch 2.7 REQUIRED)

# Match the installed LibTorch headers (PyTorch 2.13 requires C++20).
get_target_property(QWEN_CXX_STANDARD torch CXX_STANDARD)
if(NOT QWEN_CXX_STANDARD)
  set(QWEN_CXX_STANDARD 17)
endif()

# Header-only JSON parsing; pinned release and checksum, no runtime dependency.
include(FetchContent)
FetchContent_Declare(nlohmann_json
  URL https://github.com/nlohmann/json/releases/download/v3.11.3/json.tar.xz
  URL_HASH SHA256=d6c65aca6b1ed68e7a182f4757257b107ae403032760ed6ef121c9d55e81757d)
FetchContent_MakeAvailable(nlohmann_json)

add_executable(qwen_aoti main.cpp)
option(QWEN_BUILD_VALIDATION "Build the native validation driver" OFF)
option(QWEN_BUILD_BENCHMARK "Build the fixed-token native benchmark" OFF)
set(QWEN_TARGETS qwen_aoti)
if(QWEN_BUILD_VALIDATION)
  add_executable(qwen_validate tests/validate_native.cpp)
  list(APPEND QWEN_TARGETS qwen_validate)
endif()
if(QWEN_BUILD_BENCHMARK)
  add_executable(qwen_benchmark benchmark.cpp)
  list(APPEND QWEN_TARGETS qwen_benchmark)
endif()
# Optional instrumentation; the production CLI and benchmark are unchanged.
option(QWEN_BUILD_PROFILE "Build the NVTX profiling driver" OFF)
if(QWEN_BUILD_PROFILE)
  find_package(CUDAToolkit REQUIRED)
  add_executable(qwen_profile profile.cpp)
  target_link_libraries(qwen_profile PRIVATE CUDA::cudart CUDA::nvtx3)
  add_executable(qwen_graph_probe profile_graph_probe.cpp)
  target_link_libraries(qwen_graph_probe PRIVATE CUDA::cudart)
  list(APPEND QWEN_TARGETS qwen_profile qwen_graph_probe)
endif()
separate_arguments(TORCH_FLAGS NATIVE_COMMAND "${TORCH_CXX_FLAGS}")
add_library(xinfer_attention_ops SHARED attention_ops.cpp)
set_target_properties(xinfer_attention_ops PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
target_compile_features(xinfer_attention_ops PRIVATE cxx_std_${QWEN_CXX_STANDARD})
target_compile_options(xinfer_attention_ops PRIVATE ${TORCH_FLAGS} -Wall -Wextra)
target_link_libraries(xinfer_attention_ops PRIVATE ${TORCH_LIBRARIES})
foreach(target IN LISTS QWEN_TARGETS)
  target_compile_features(${target} PRIVATE cxx_std_${QWEN_CXX_STANDARD})
  target_compile_options(${target} PRIVATE ${TORCH_FLAGS} -Wall -Wextra)
  target_link_libraries(${target} PRIVATE ${TORCH_LIBRARIES} nlohmann_json::nlohmann_json xinfer_attention_ops)
endforeach()

option(QWEN_BUILD_DELTANET "Build the isolated Qwen3.5 DeltaNet probe" OFF)
if(QWEN_BUILD_DELTANET)
  add_subdirectory(qwen35)
endif()
